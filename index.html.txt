    const STORAGE_KEYS = {
      section: "bunko_current_section",
      bookmark: "bunko_bookmark_section",
      theme: "bunko_theme",
      writing: "bunko_writing",
      fontSize: "bunko_font_size"
    };

    const coverScreen = document.getElementById("coverScreen");
    const readerScreen = document.getElementById("readerScreen");
    const pageBody = document.getElementById("pageBody");
    const sectionTitle = document.getElementById("sectionTitle");
    const sectionNo = document.getElementById("sectionNo");
    const pageIndicator = document.getElementById("pageIndicator");
    const headerTitle = document.getElementById("headerTitle");
    const readingTheme = document.getElementById("readingTheme");
    const tocList = document.getElementById("tocList");
    const overlay = document.getElementById("overlay");
    const drawer = document.getElementById("drawer");

    let currentSection = parseInt(localStorage.getItem(STORAGE_KEYS.section) || "0", 10);
    let currentTheme = localStorage.getItem(STORAGE_KEYS.theme) || "light";
    let currentWriting = localStorage.getItem(STORAGE_KEYS.writing) || "horizontal";
    let currentFontSize = parseInt(localStorage.getItem(STORAGE_KEYS.fontSize) || "18", 10);

    function saveCurrentSection() {
      localStorage.setItem(STORAGE_KEYS.section, String(currentSection));
    }

    function applyTheme(theme) {
      document.body.classList.remove("theme-light", "theme-dark", "theme-midnight");
      document.body.classList.add(`theme-${theme}`);
      currentTheme = theme;
      localStorage.setItem(STORAGE_KEYS.theme, theme);
      readingTheme.textContent = theme.toUpperCase();
      document.querySelectorAll("#themeButtons button").forEach(btn => {
        btn.classList.toggle("active", btn.dataset.theme === theme);
      });
    }

    function applyWritingMode(mode) {
      currentWriting = mode;
      localStorage.setItem(STORAGE_KEYS.writing, mode);
      pageBody.classList.toggle("vertical", mode === "vertical");
      document.querySelectorAll("#writingButtons button").forEach(btn => {
        btn.classList.toggle("active", btn.dataset.writing === mode);
      });
      document.getElementById("toggleVerticalBtn").textContent = mode === "vertical" ? "横" : "縦";
    }

    function applyFontSize(size) {
      currentFontSize = Math.min(26, Math.max(14, size));
      document.documentElement.style.setProperty("--font-size", currentFontSize + "px");
      localStorage.setItem(STORAGE_KEYS.fontSize, String(currentFontSize));
    }

    function renderTOC() {
      tocList.innerHTML = "";
      STORY.forEach((section, index) => {
        const btn = document.createElement("button");
        btn.className = "toc-item" + (index === currentSection ? " active" : "");
        btn.innerHTML = `<strong>${index + 1}. ${section.title}</strong>`;
        btn.addEventListener("click", () => {
          currentSection = index;
          renderSection();
          closeDrawer();
          openReader();
        });
        tocList.appendChild(btn);
      });
    }

    function renderSection() {
      const section = STORY[currentSection];
      sectionTitle.textContent = section.title;
      sectionNo.textContent = `SECTION ${currentSection + 1}`;
      headerTitle.textContent = `今日、学校休みかな — ${section.title}`;
      pageIndicator.textContent = `${currentSection + 1} / ${STORY.length}`;

      pageBody.innerHTML = "";
      section.content.forEach(paragraph => {
        const p = document.createElement("p");
        p.textContent = paragraph;
        pageBody.appendChild(p);
      });

      document.getElementById("prevBtn").disabled = currentSection === 0;
      document.getElementById("nextBtn").disabled = currentSection === STORY.length - 1;
      document.getElementById("prevBtn").style.opacity = currentSection === 0 ? ".45" : "1";
      document.getElementById("nextBtn").style.opacity = currentSection === STORY.length - 1 ? ".45" : "1";

      saveCurrentSection();
      renderTOC();
      pageBody.scrollTop = 0;
      pageBody.scrollLeft = 0;
    }

    function openReader() {
      coverScreen.classList.remove("active");
      readerScreen.classList.add("active");
    }

    function openCover() {
      readerScreen.classList.remove("active");
      coverScreen.classList.add("active");
    }

    function openDrawer() {
      overlay.classList.add("show");
      drawer.classList.add("show");
    }

    function closeDrawer() {
      overlay.classList.remove("show");
      drawer.classList.remove("show");
    }

    function saveBookmark() {
      localStorage.setItem(STORAGE_KEYS.bookmark, String(currentSection));
      alert("しおりを保存しました。");
    }

    function resumeFromBookmark() {
      const bookmark = localStorage.getItem(STORAGE_KEYS.bookmark);
      if (bookmark !== null) {
        currentSection = parseInt(bookmark, 10);
      }
      renderSection();
      openReader();
    }

    document.getElementById("startReadingBtn").addEventListener("click", () => {
      currentSection = 0;
      renderSection();
      openReader();
    });

    document.getElementById("resumeReadingBtn").addEventListener("click", () => {
      resumeFromBookmark();
    });

    document.getElementById("backToCoverBtn").addEventListener("click", openCover);
    document.getElementById("openDrawerBtn").addEventListener("click", openDrawer);
    overlay.addEventListener("click", closeDrawer);

    document.getElementById("prevBtn").addEventListener("click", () => {
      if (currentSection > 0) {
        currentSection--;
        renderSection();
      }
    });

    document.getElementById("nextBtn").addEventListener("click", () => {
      if (currentSection < STORY.length - 1) {
        currentSection++;
        renderSection();
      }
    });

    document.getElementById("bookmarkBtn").addEventListener("click", saveBookmark);

    document.getElementById("smallerFontBtn").addEventListener("click", () => {
      applyFontSize(currentFontSize - 1);
    });

    document.getElementById("largerFontBtn").addEventListener("click", () => {
      applyFontSize(currentFontSize + 1);
    });

    document.getElementById("drawerSmallerBtn").addEventListener("click", () => {
      applyFontSize(currentFontSize - 1);
    });

    document.getElementById("drawerLargerBtn").addEventListener("click", () => {
      applyFontSize(currentFontSize + 1);
    });

    document.getElementById("toggleVerticalBtn").addEventListener("click", () => {
      applyWritingMode(currentWriting === "vertical" ? "horizontal" : "vertical");
    });

    document.querySelectorAll("#themeButtons button").forEach(btn => {
      btn.addEventListener("click", () => applyTheme(btn.dataset.theme));
    });

    document.querySelectorAll("#writingButtons button").forEach(btn => {
      btn.addEventListener("click", () => applyWritingMode(btn.dataset.writing));
    });

    // swipe
    let touchStartX = 0;
    let touchStartY = 0;

    pageBody.addEventListener("touchstart", (e) => {
      const t = e.changedTouches[0];
      touchStartX = t.clientX;
      touchStartY = t.clientY;
    }, { passive: true });

    pageBody.addEventListener("touchend", (e) => {
      const t = e.changedTouches[0];
      const dx = t.clientX - touchStartX;
      const dy = t.clientY - touchStartY;

      if (Math.abs(dx) > 70 && Math.abs(dx) > Math.abs(dy)) {
        if (dx < 0 && currentSection < STORY.length - 1) {
          currentSection++;
          renderSection();
        } else if (dx > 0 && currentSection > 0) {
          currentSection--;
          renderSection();
        }
      }
    }, { passive: true });

    // keyboard
    window.addEventListener("keydown", (e) => {
      if (e.key === "ArrowRight") {
        if (currentSection < STORY.length - 1) {
          currentSection++;
          renderSection();
        }
      } else if (e.key === "ArrowLeft") {
        if (currentSection > 0) {
          currentSection--;
          renderSection();
        }
      } else if (e.key === "Escape") {
        closeDrawer();
      }
    });

    // init
    applyTheme(currentTheme);
    applyWritingMode(currentWriting);
    applyFontSize(currentFontSize);
    renderTOC();
    renderSection();

    // optional service worker
    if ("serviceWorker" in navigator) {
      window.addEventListener("load", () => {
        navigator.serviceWorker.register("./sw.js").catch(() => {});
      });
    }
  </script>
